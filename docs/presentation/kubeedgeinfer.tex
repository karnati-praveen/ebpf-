      \kedot{3}\ \kehead{What existing frameworks do}\par\vspace{0.3em}

      {\scriptsize\color{keteal}\bfseries IEEE IoT-J, 2024}\newline
      {\tiny\ttfamily arXiv:2405.14371} &
      \kesub{Joint device selection and layer partition by dynamic programming

      {\scriptsize\color{keteal}\bfseries IEEE INFOCOM, 2024}\newline
      {\tiny\ttfamily arXiv:2405.17245} &
      \kesub{Hybrid tensor/sequence parallelism for in-situ Transformer

      compute/communication overlap. Up to 2.5$\times$ lower latency.} &
      \kelim{Planning is a one-shot offline step; no runtime re-planning under
      drift.}\\[0.5em]
    \kehead{Helix}\newline
      {\scriptsize\color{keteal}\bfseries ACM ASPLOS, 2025}\newline
      {\tiny\ttfamily arXiv:2406.01566} &
      \kesub{Models heterogeneous GPU serving as max-flow on a weighted graph
      and solves placement and request scheduling jointly by MILP. Up to
      3.3$\times$ throughput on 24--42 node clusters.} &
      \kelim{Datacenter-scale GPUs; MILP is too heavy for a seconds-scale
      control loop.}\\[0.5em]
    \kehead{TPI-LLM}\newline
      {\scriptsize\color{keteal}\bfseries arXiv, 2024}\newline
      {\tiny\ttfamily arXiv:2410.00531} &
      \kesub{Argues tensor parallelism beats pipeline on low-resource devices;
      a sliding-window memory scheduler and star-based allreduce keep 70B-scale
      weights moving through small RAM.} &
      \kelim{Optimises memory and collectives, not adaptation to node
      degradation.}\\[0.5em]
    \kehead{prima.cpp}\newline
      {\scriptsize\color{keteal}\bfseries arXiv, 2025}\newline
      {\tiny\ttfamily arXiv:2504.08791} &
      \kesub{30--70B inference on real home clusters: the Halda scheduler
      co-optimises CPU/GPU workload and device selection; pipelined-ring
      parallelism hides disk I/O. 5--17$\times$ lower TPOT than llama.cpp.} &
      \kelim{Scheduling is decided at load time from static device
      capability.}\\

  \end{tabular}

% =============================================================== 6. Problem
{\kedark
\begin{frame}[shrink=20]
  \vspace{0.6em}
  {\scriptsize\bfseries\color{keamber}PROBLEM STATEMENT}\par\vspace{0.7em}
  {\headfont\large\color{white}
   Given $N$ transformer layers and $K$ heterogeneous edge nodes whose compute
   speed, link latency and liveness all vary during inference, continuously
   maintain a contiguous layer assignment that minimises the bottleneck stage
   --- without restarting the pipeline.}\par\vspace{1.2em}
  \begin{columns}[T]
  \begin{column}{0.62\textwidth}\footnotesize
    \kedot{1}\ {\bfseries\color{white}Measure}\par
    {\scriptsize\color{white!60!keink}Obtain per-flow latency and per-node
     thermal state cheaply and without instrumenting the inference
     code.}\par\vspace{0.6em}
    \kedot[keteal]{2}\ {\bfseries\color{white}Decide}\par
    {\scriptsize\color{white!60!keink}Solve the placement exactly at telemetry
     rate, and suppress oscillation when the measurement is merely
     noisy.}\par\vspace{0.6em}
    \kedot{3}\ {\bfseries\color{white}Act}\par
    {\scriptsize\color{white!60!keink}Move layer ranges between live workers
     without dropping in-flight requests or restarting
     pods.}\par\vspace{0.6em}
    \kedot[keteal]{4}\ {\bfseries\color{white}Heal}\par
    {\scriptsize\color{white!60!keink}Detect node loss and redistribute its
     layers across the survivors automatically.}
  \end{column}
  \begin{column}{0.34\textwidth}
    \colorbox{keink2}{\begin{minipage}{\dimexpr\textwidth-14pt\relax}
      \vspace{0.4em}
      {\scriptsize\bfseries\color{keamber}CONSTRAINT}\par\vspace{0.6em}
      {\footnotesize\color{white!78!keink}Commodity Linux only.\par\vspace{0.5em}
       No datacenter interconnect, no vendor telemetry agent, no change to the
       model code, and no assumption that a node stays alive.}
      \vspace{0.5em}
    \end{minipage}}
  \end{column}
  \end{columns}
\end{frame}}

% =============================================================== 7. Contributions
\begin{frame}[shrink=20]
  \framesubtitle{Contributions}
  \frametitle{What is new here}
  \footnotesize
  \begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \kecard{\textwidth}{\kedot{1}\ \kehead{Kernel telemetry as a scheduling input}\par
      \vspace{0.25em}\kesub{CO-RE eBPF programs (\texttt{tp\_btf/tcp\_probe},
      \texttt{fentry/tcp\_sendmsg}) feed per-flow smoothed RTT straight into a
      model-placement decision. Prior edge-LLM systems use user-space probes or
      offline profiles; none drives the partitioner from the kernel's own TCP
      state.}}\vspace{0.35em}
    \kecard{\textwidth}{\kedot{2}\ \kehead{A closed loop, not a one-shot plan}\par
      \vspace{0.25em}\kesub{WATCH--DECIDE--ACT--HEAL runs every 2\,s for the
      lifetime of the deployment. The split is a continuously maintained control
      variable rather than a deployment-time constant.}}\vspace{0.35em}
    \kecard{\textwidth}{\kedot{3}\ \kehead{Hysteresis-gated exact optimisation}\par
      \vspace{0.25em}\kesub{The $O(N^2K)$ DP is exact for each snapshot; a 15\%
      improvement threshold plus a 30\,s cooldown converts it into a stable
      controller. One gate handles thermal, network and failure triggers
      alike.}}
  \end{column}
  \begin{column}{0.48\textwidth}
    \kecard[kemist2]{\textwidth}{\kedot[keteal]{4}\ \kehead{Restart-free, generation-fenced reassignment}\par
      \vspace{0.25em}\kesub{Layer ranges move over gRPC while requests are in
      flight. Every assignment carries a generation; workers reject stale
      forwards and the stateless design makes context replay trivially
      correct.}}\vspace{0.35em}
    \kecard[kemist2]{\textwidth}{\kedot[keteal]{5}\ \kehead{An ablation that isolates continuous telemetry}\par
      \vspace{0.25em}\kesub{A profile-once mode reuses the same DP and apply
      path but freezes its inputs, approximating offline-profiling systems --
      so the measured gain is attributable to live telemetry, not to a better
      solver.}}\vspace{0.35em}
    \colorbox{keink}{\begin{minipage}{\dimexpr\textwidth-14pt\relax}
      \vspace{0.3em}
      {\scriptsize\bfseries\color{keamber}CORE NOVELTY CLAIM}\par\vspace{0.4em}
      {\footnotesize\color{white!80!keink}Not a better partitioning algorithm
       --- the same classical DP the literature already uses, placed inside a
       kernel-measured feedback loop that never stops running.}
      \vspace{0.3em}
    \end{minipage}}
  \end{column}
  \end{columns}
  \kefoot{Items 1--3 are the primary novelty claims.}
\end{frame}

% =============================================================== 9. Formulation
\begin{frame}[shrink=20]
  \framesubtitle{Proposed work}
  \frametitle{Formulation and stability}
  \begin{columns}[T]
  \begin{column}{0.47\textwidth}\footnotesize
    \kecard{\textwidth}{%
      {\bfseries\color{keamber}Objective}\par\vspace{0.4em}
      \[\min\ \max_{i\in 1..K}\ \mathrm{cost}(i)\]
      \vspace{-0.8em}
      \[\mathrm{cost}(i)=\frac{\ell_i \cdot t_{\text{layer}}}{s_i}
                        + \mathrm{link}_{i-1}\]
      \vspace{-0.4em}
      \kesub{subject to contiguous, non-overlapping ranges covering all $N$
      layers. An empty stage costs nothing --- so the DP may bypass a worker
      entirely when its link has degraded past the value of its
      compute.}}\vspace{0.4em}
    \kecard{\textwidth}{%
      {\bfseries\color{keteal}Solution}\par\vspace{0.4em}
      \[f[j][w]=\min_{\text{split}}\ \max\bigl(f[\text{split}][w{-}1],\,
        \mathrm{cost}(w,\,j-\text{split})\bigr)\]
      \vspace{-0.4em}
      \kesub{The classical linear-partition recurrence: minimal bottleneck for
      the first $j$ layers over the first $w$ workers. $O(N^2K)$ --- about 1.7k
      operations for $N{=}12$, $K{=}3$, so re-solving at 0.5\,Hz is free.
      Verified against brute force on 200 randomised instances.}}
  \end{column}
  \begin{column}{0.49\textwidth}
    \colorbox{keink}{\begin{minipage}{\dimexpr\textwidth-14pt\relax}\footnotesize
      \vspace{0.4em}
      {\bfseries\color{keamber}Why a bare optimiser is not a controller}\par
      \vspace{0.4em}
      {\scriptsize\color{white!68!keink}Telemetry is noisy. Re-applying the
       argmin on every 2\,s tick would thrash the pipeline for sub-millisecond
       gains. Three rules turn the optimiser into a stable
       controller:}\par\vspace{0.8em}
      {\color{white}\bfseries\scriptsize\textcolor{keteal}{$\bullet$}\ Improvement threshold}\par
      {\scriptsize\color{white!60!keink}Apply only if the new split beats the
       current one, re-evaluated under current telemetry, by
       15\%.}\par\vspace{0.5em}
      {\color{white}\bfseries\scriptsize\textcolor{keteal}{$\bullet$}\ Cooldown}\par
      {\scriptsize\color{white!60!keink}At most one change per 30\,s, so a
       transient spike cannot start an oscillation.}\par\vspace{0.5em}
      {\color{white}\bfseries\scriptsize\textcolor{keamber}{$\bullet$}\ Membership override}\par
      {\scriptsize\color{white!60!keink}A changed worker set bypasses both gates
       and repartitions immediately --- correctness beats
       stability.}\par\vspace{0.8em}
      {\tiny\ttfamily\color{kemuted}Both thresholds are tunable live via
       IMPROVEMENT\_FRAC and COOLDOWN\_S.}
      \vspace{0.4em}
    \end{minipage}}
  \end{column}
  \end{columns}
\end{frame}

% =============================================================== 14. Conclusion
{\kedark
\begin{frame}[shrink=20]
  \vspace{0.5em}
  {\scriptsize\bfseries\color{keamber}CONCLUSION}\par\vspace{0.7em}
  {\headfont\Large\bfseries\color{white}
   The split point should be a control variable, not a deployment
   constant.}\par\vspace{0.9em}
  {\footnotesize\color{white!62!keink}
   KubeEdgeInfer keeps a classical exact partitioner permanently in the loop,
   fed by kernel-measured latency and real thermal state, and applies its
   decisions to a running pipeline without a restart. On a heterogeneous edge
   cluster that is the difference between a plan that was right once and a plan
   that stays right.}\par\vspace{1.1em}
  \begin{columns}[T]
  \begin{column}{0.60\textwidth}\footnotesize
    \kedot{1}\ {\bfseries\color{white}Exactness is cheap}\par
    {\scriptsize\color{white!58!keink}The DP costs microseconds; there is no
     reason to solve it only once.}\par\vspace{0.5em}
    \kedot[keteal]{2}\ {\bfseries\color{white}Hysteresis is what makes it usable}\par
    {\scriptsize\color{white!58!keink}One improvement threshold and one cooldown
     tame thermal, network and failure triggers alike.}\par\vspace{0.5em}
    \kedot[keteal]{3}\ {\bfseries\color{white}Kernel telemetry needs no instrumentation}\par
    {\scriptsize\color{white!58!keink}eBPF measures the real transport, not what
     the application thinks it sent.}
  \end{column}
  \begin{column}{0.36\textwidth}
    \colorbox{keink2}{\begin{minipage}{\dimexpr\textwidth-14pt\relax}
